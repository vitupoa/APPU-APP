// Helper to map replyToUser to camelCase
const mapReplyToUser = (user: any) => {
  if (!user) return null;
  return {
    id: user.id,
    username: user.username,
    displayName: user.display_name,
    avatarUrl: user.avatar_url,
    isVerified: user.is_verified
  };
};
